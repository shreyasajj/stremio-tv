cmake_minimum_required(VERSION 3.21)
project(StremioTVNative VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 6.4 REQUIRED COMPONENTS Quick Qml OpenGL)
find_package(PkgConfig REQUIRED)
pkg_check_modules(MPV REQUIRED IMPORTED_TARGET mpv)
qt_add_executable(stremio-tv-native main.cpp mpvitem.cpp mpvitem.h)
qt_add_qml_module(stremio-tv-native URI StremioTV VERSION 1.0 QML_FILES Main.qml)
target_link_libraries(stremio-tv-native PRIVATE Qt6::Quick Qt6::Qml Qt6::OpenGL PkgConfig::MPV)
install(TARGETS stremio-tv-native RUNTIME DESTINATION bin)
